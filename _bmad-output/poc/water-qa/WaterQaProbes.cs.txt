// w-qa temporary probes for the water verdict (slot #12). Removed after the slot; results go to TestResults/water-qa/.
//   unity run <project> -- -executeMethod WaterQaProbes.RunAll
using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using System.Text;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using WashedAshore.World;

public static class WaterQaProbes
{
    static string Out => Path.Combine(BellsBendData.ProjectRoot, "TestResults", "water-qa");

    public static void RunProbeB()
    {
        Directory.CreateDirectory(Out);
        string r; int code = 0;
        try { r = ProbeB(); } catch (Exception e) { r = "FAILED: " + e; code = 1; }
        File.WriteAllText(Path.Combine(Out, "qa-probe-b-rerun.txt"), $"W3 probe (b) rerun {DateTime.UtcNow:yyyy-MM-ddTHH:mm:ssZ}\n{r}\ndone\n");
        EditorApplication.Exit(code);
    }

    public static void RunAll()
    {
        Directory.CreateDirectory(Out);
        var log = new StringBuilder($"WaterQaProbes {DateTime.UtcNow:yyyy-MM-ddTHH:mm:ssZ} batch={Application.isBatchMode}\n");
        int code = 0;
        foreach (var step in new (string name, Func<string> run)[] { ("W3-probe-b", ProbeB), ("AF1-water-sky", Af1), ("west-channel-shot", WestChannelShot) })
        {
            try { log.AppendLine($"== {step.name}\n{step.run()}"); }
            catch (Exception e) { code = 1; log.AppendLine($"== {step.name} FAILED: {e}"); }
            File.WriteAllText(Path.Combine(Out, "qa-probes.txt"), log.ToString());
        }
        log.AppendLine("done");
        File.WriteAllText(Path.Combine(Out, "qa-probes.txt"), log.ToString());
        EditorApplication.Exit(code);
    }

    static string Sha(string rel) => File.Exists(Path.Combine(BellsBendData.ProjectRoot, rel))
        ? BitConverter.ToString(SHA256.Create().ComputeHash(File.ReadAllBytes(Path.Combine(BellsBendData.ProjectRoot, rel)))).Replace("-", "").Substring(0, 16).ToLowerInvariant()
        : "missing";

    static List<string> Watched()
    {
        var files = new List<string> { BellsBendLevel.ScenePath, BellsBendData.MapConfigPath };
        files.AddRange(Directory.GetFiles(Path.Combine(BellsBendData.ProjectRoot, "Assets/World/BellsBend/Tiles"), "*.asset")
            .Select(f => "Assets/World/BellsBend/Tiles/" + Path.GetFileName(f)).OrderBy(f => f));
        return files;
    }

    // W3 probe (b): in-memory MapConfig clone with northLineZ + 100. Guard alone first; BuildAll only if Guard threw.
    static string ProbeB()
    {
        var sb = new StringBuilder();
        var files = Watched();
        var before = files.ToDictionary(f => f, Sha);
        sb.AppendLine($"watched files: {files.Count} (World.unity, MapConfig.asset, {files.Count - 2} tiles)");
        // Open the scene first: BuildAll's OpenScene would otherwise unload the unreferenced in-memory clone.
        if (EditorSceneManager.GetActiveScene().path != BellsBendLevel.ScenePath) EditorSceneManager.OpenScene(BellsBendLevel.ScenePath);
        var c = UnityEngine.Object.Instantiate(BellsBendData.LoadConfig());
        c.hideFlags = HideFlags.DontUnloadUnusedAsset;
        c.northLineZ += 100f;
        bool guardThrew = false;
        try { BellsBendWater.Guard(c, BellsBendData.LoadManifest(c)); }
        catch (WaterBuildException e) { guardThrew = true; sb.AppendLine("Guard threw WaterBuildException: " + e.Message); }
        if (!guardThrew) { sb.AppendLine("RESULT: FAIL - Guard did not throw; BuildAll NOT called"); return sb.ToString(); }
        string buildAllResult;
        try { var r = BellsBendLevel.BuildAll(c, false); buildAllResult = "returned without throwing: " + (r.Length > 300 ? r.Substring(0, 300) : r); }
        catch (WaterBuildException e) { buildAllResult = "threw WaterBuildException: " + e.Message; }
        catch (Exception e) { buildAllResult = "threw " + e.GetType().Name + ": " + e.Message; }
        sb.AppendLine("BuildAll(clone, false) " + buildAllResult);
        UnityEngine.Object.DestroyImmediate(c);
        var after = files.ToDictionary(f => f, Sha);
        var changed = files.Where(f => before[f] != after[f]).ToList();
        sb.AppendLine($"files changed: {changed.Count} {string.Join(", ", changed)}");
        bool pass = buildAllResult.StartsWith("threw WaterBuildException") && buildAllResult.Contains("north line moved") && changed.Count == 0;
        sb.AppendLine("RESULT: " + (pass ? "PASS" : "FAIL"));
        // Discard any in-memory scene state the attempt may have left.
        EditorSceneManager.OpenScene(BellsBendLevel.ScenePath, OpenSceneMode.Single);
        return sb.ToString();
    }

    // AF1 (water pod scope): content fingerprint of the Water and Sky roots + generated assets, rebuild by script, compare.
    static string Fingerprint()
    {
        EditorSceneManager.OpenScene(BellsBendLevel.ScenePath, OpenSceneMode.Single);
        var sb = new StringBuilder();
        sb.AppendLine($"RenderSettings fog={RenderSettings.fogColor} mode={RenderSettings.fogMode} {RenderSettings.fogStartDistance}-{RenderSettings.fogEndDistance} density={RenderSettings.fogDensity} skybox={(RenderSettings.skybox ? AssetDatabase.GetAssetPath(RenderSettings.skybox) : "none")}");
        foreach (var rootName in new[] { "Water", "Sky" })
        {
            var roots = EditorSceneManager.GetActiveScene().GetRootGameObjects().Where(g => g.name == rootName).ToArray();
            sb.AppendLine($"[{rootName}] roots={roots.Length}");
            foreach (var t in roots.SelectMany(r => r.GetComponentsInChildren<Transform>(true)).OrderBy(t => TPath(t)))
            {
                sb.AppendLine($"{TPath(t)} layer={t.gameObject.layer} active={t.gameObject.activeSelf} pos={t.position.ToString("F4")} rot={t.rotation.eulerAngles.ToString("F3")} scl={t.lossyScale.ToString("F4")}");
                foreach (var comp in t.GetComponents<Component>())
                {
                    if (comp is Transform) continue;
                    string json = comp ? EditorJsonUtility.ToJson(comp) : "null";
                    sb.AppendLine($"  {comp?.GetType().Name}: {json}");
                }
            }
        }
        foreach (var a in new[] { "Assets/World/BellsBend/Water/WaterTile.asset", "Assets/World/BellsBend/Water/WaterMaps.png",
                     "Assets/World/BellsBend/Water/WaterRipple.png", "Assets/World/BellsBend/Water/BellsBendWater.mat",
                     "Assets/World/BellsBend/Sky/BellsBendSky.mat", "Assets/World/Water/WaterMotion.asset" })
            sb.AppendLine($"{a} {Sha(a)}");
        return sb.ToString();
    }

    static string TPath(Transform t) => t.parent ? TPath(t.parent) + "/" + t.name : t.name;

    static string Af1()
    {
        var sb = new StringBuilder();
        var f0 = Fingerprint();
        File.WriteAllText(System.IO.Path.Combine(Out, "af1-F0.txt"), f0);
        sb.AppendLine("water: " + BellsBendWater.Build().Replace("\n", " | "));
        sb.AppendLine("sky: " + BellsBendSky.Build().Replace("\n", " | "));
        var f1 = Fingerprint();
        File.WriteAllText(System.IO.Path.Combine(Out, "af1-F1.txt"), f1);
        var a = f0.Split('\n'); var b = f1.Split('\n');
        var diffs = new List<string>();
        for (int i = 0; i < Math.Max(a.Length, b.Length); i++)
        {
            var x = i < a.Length ? a[i] : "<none>"; var y = i < b.Length ? b[i] : "<none>";
            if (x != y) diffs.Add($"line {i}: F0 {Trim(x)} | F1 {Trim(y)}");
        }
        sb.AppendLine($"F0 lines {a.Length}, F1 lines {b.Length}, differing lines {diffs.Count}");
        foreach (var d in diffs.Take(20)) sb.AppendLine(d);
        sb.AppendLine("RESULT: " + (diffs.Count == 0 ? "PASS (F1 == F0)" : "DIFF"));
        return sb.ToString();
    }

    static string Trim(string s) => s.Length > 200 ? s.Substring(0, 200) + "..." : s;

    // W5 (c): a west-channel view north of the line, to pair with w-artist's east-channel shot. Fog on, no player.
    static string WestChannelShot()
    {
        EditorSceneManager.OpenScene(BellsBendLevel.ScenePath, OpenSceneMode.Single);
        var cfg = BellsBendData.LoadConfig();
        var shots = new[]
        {
            ("w5-north-line-west-channel.png", new Vector3(-1440f, cfg.WaterLevelY + 8f, cfg.northLineZ - 40f), new Vector3(-1470f, cfg.WaterLevelY, cfg.northLineZ + 120f)),
        };
        var sb = new StringBuilder();
        foreach (var (file, eye, look) in shots)
        {
            const int W = 1920, H = 1080;
            var go = new GameObject("_QaCam") { hideFlags = HideFlags.HideAndDontSave };
            var cam = go.AddComponent<Camera>();
            cam.transform.position = eye; cam.transform.LookAt(look);
            cam.fieldOfView = 60f; cam.nearClipPlane = 0.3f; cam.farClipPlane = 3000f;
            var rt = RenderTexture.GetTemporary(W, H, 24, RenderTextureFormat.ARGB32);
            cam.targetTexture = rt; cam.Render();
            var prev = RenderTexture.active; RenderTexture.active = rt;
            var tex = new Texture2D(W, H, TextureFormat.RGB24, false);
            tex.ReadPixels(new Rect(0, 0, W, H), 0, 0); tex.Apply();
            RenderTexture.active = prev; cam.targetTexture = null; RenderTexture.ReleaseTemporary(rt);
            var dir = System.IO.Path.Combine(BellsBendData.PocDir, "water-qa");
            Directory.CreateDirectory(dir);
            File.WriteAllBytes(System.IO.Path.Combine(dir, file), tex.EncodeToPNG());
            UnityEngine.Object.DestroyImmediate(tex); UnityEngine.Object.DestroyImmediate(go);
            sb.AppendLine($"{file}: eye {eye} -> {look} (RGB24, opaque)");
        }
        return sb.ToString();
    }
}
